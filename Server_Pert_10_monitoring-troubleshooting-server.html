2 Toolkit CLI Monitoring Server Enterprise

# ───── CPU & PROSES ───── htop # Interactive process viewer (color-coded) ps aux --sort=-%cpu | head -10 # Top 10 proses terpakai CPU terbanyak # ───── MEMORI ───── free -h # Tampil RAM & Swap dalam human-readable format vmstat 1 5 # Statistik virtual memory setiap 1 detik, 5 kali # ───── DISK ───── df -h # Penggunaan disk semua partisi du -sh /var/log/* # Ukuran setiap file/folder di /var/log iostat -xz 1 # I/O statistics per disk device # ───── JARINGAN & PORT ───── ss -tulnp # Semua socket TCP/UDP yang LISTEN beserta PID ip -s link # Statistik paket per network interface

3 Analisis Log dengan journalctl & Syslog

# Lihat semua log sistem sejak boot ini sudo journalctl -b # Filter log satu layanan spesifik (misal: apache2) sudo journalctl -u apache2.service -f # Tampil log 1 jam terakhir semua layanan sudo journalctl --since "1 hour ago" # Filter berdasarkan tingkat severity (err, warning, crit) sudo journalctl -p err -b # Log autentikasi (SSH, sudo, login gagal) sudo tail -f /var/log/auth.log | grep "Failed" # Log kernel (dmesg — error hardware, driver) dmesg | grep -E "error|fail|warn" -i
