2 Konfigurasi Lengkap Virtual Host Apache2 & Nginx

# 1. /etc/apache2/sites-available/portal.conf — Apache2 Virtual Host 
<VirtualHost *:80>
    ServerName portal.smk.lan
    ServerAlias www.portal.smk.lan
    ServerAdmin webmaster@smk.lan
    DocumentRoot /var/www/portal

    <Directory /var/www/portal>
        Options -Indexes +FollowSymLinks
        AllowOverride All
        Require all granted
    </Directory>

    ErrorLog ${APACHE_LOG_DIR}/portal_error.log
    CustomLog ${APACHE_LOG_DIR}/portal_access.log combined
</VirtualHost>
# 2. /etc/nginx/sites-available/portal.conf — Nginx Server Block 
server {
    listen 80;
    server_name portal.smk.lan www.portal.smk.lan;
    root /var/www/portal;
    index index.php index.html;

    location / {
        try_files $uri $uri/ =404;
    }

    location ~ \.php$ {
        include snippets/fastcgi-php.conf;
        fastcgi_pass unix:/var/run/php/php8.1-fpm.sock;
    }
}
