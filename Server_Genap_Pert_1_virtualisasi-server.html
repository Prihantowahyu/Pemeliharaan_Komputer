2 Manajemen VM via CLI VBoxManage & Proxmox

# VirtualBox: Daftar semua VM yang terdaftar VBoxManage list vms # Buat VM baru: Ubuntu Server 22.04, 2 CPU, 2GB RAM, 20GB Disk VBoxManage createvm --name "UbuntuServer" --ostype Ubuntu_64 --register VBoxManage modifyvm "UbuntuServer" --cpus 2 --memory 2048 --vram 16 VBoxManage createhd --filename "UbuntuServer.vdi" --size 20480 VBoxManage storagectl "UbuntuServer" --name "SATA" --add sata VBoxManage storageattach "UbuntuServer" --storagectl "SATA" --port 0 --device 0 --type hdd --medium "UbuntuServer.vdi" # Start/Stop/Suspend VM VBoxManage startvm "UbuntuServer" --type headless VBoxManage controlvm "UbuntuServer" poweroff
# Proxmox VE: Manajemen VM via pvesh CLI (di host Proxmox) pvesh get /nodes/pve/qemu # Daftar semua VM qm start 100 # Start VM dengan ID 100 qm snapshot 100 snap-before-update # Buat snapshot VM sebelum update qm rollback 100 snap-before-update # Rollback ke snapshot jika terjadi masalah
