2 Skrip Automasi Backup Bash & Format Penjadwalan Crontab

# 1. /usr/local/bin/backup_server.sh — Skrip Bash Backup Harian 
#!/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
TIMESTAMP=$(date +"%Y%m%d_%H%M%S")
BACKUP_DIR="/backup/local"
REMOTE_HOST="backupuser@192.168.10.200"

mkdir -p ${BACKUP_DIR}

# 1. Backup Database MySQL
mysqldump -u root -p'PasswordDB123' --all-databases | gzip > ${BACKUP_DIR}/db_full_${TIMESTAMP}.sql.gz

# 2. Sinkronisasi Data Web & Berkas ke Server Remote (3-2-1 Strategy)
rsync -avzP --delete /var/www/html/ ${REMOTE_HOST}:/backup/storage/web_html/

# 3. Hapus salinan lokal yang lebih tua dari 7 hari (Rotasi Penyimpanan)
find ${BACKUP_DIR} -type f -name "*.sql.gz" -mtime +7 -exec rm -f {} \;
# 2. Format Crontab (crontab -e) — Jalankan Setiap Jam 02:00 Dini Hari 
# ┌───────────── Menit (0 - 59)
# │ ┌────────────── Jam (0 - 23)
# │ │ ┌─────────────── Hari dari Bulan (1 - 31)
# │ │ │ ┌──────────────── Bulan (1 - 12)
# │ │ │ │ ┌───────────────── Hari dari Minggu (0 - 6, 0 = Minggu)
# │ │ │ │ │
  0 2 * * * /bin/bash /usr/local/bin/backup_server.sh >> /var/log/backup_cron.log 2>&1
