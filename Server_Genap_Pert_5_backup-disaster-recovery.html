2 Implementasi rsync + cron Otomatis

# rsync — Sinkronisasi inkremental lokal # -a: archive mode (preserve permissions, timestamps), -v: verbose, -z: compress rsync -avz --delete /var/www/html/ /mnt/backup/www/ # rsync ke remote server via SSH (backup offsite) rsync -avz -e "ssh -p 2222 -i ~/.ssh/id_smk" /var/www/html/ backup@192.168.10.2:/backup/www/ # Backup database MySQL otomatis mysqldump -u root -p'P@ssw0rd' --all-databases | gzip > /mnt/backup/db/all_db_$(date +%Y%m%d).sql.gz
# Cron Job — edit dengan: crontab -e # Format: menit jam hari bulan hari_minggu perintah # Backup website setiap hari pukul 02:00 dini hari 0 2 * * * rsync -az --delete /var/www/html/ /mnt/backup/www/ # Backup database setiap hari pukul 01:30 30 1 * * * mysqldump -u root -pP@ssw0rd --all-databases | gzip > /mnt/backup/db/db_$(date +\%Y\%m\%d).sql.gz # Hapus backup lebih dari 30 hari — jaga agar storage tidak penuh 0 3 * * 0 find /mnt/backup/ -name "*.sql.gz" -mtime +30 -delete
