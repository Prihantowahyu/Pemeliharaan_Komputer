2 Konfigurasi Lengkap Nginx Reverse Proxy Load Balancer

# File: /etc/nginx/sites-available/load_balancer.conf 
# 1. Definisi Pool Server Backend (Cluster Upstream)
upstream web_cluster {
    # Algoritma Default: Round Robin (Bisa diganti 'least_conn;' atau 'ip_hash;')
    least_conn;

    server 192.168.10.11:80 max_fails=2 fail_timeout=10s weight=2;
    server 192.168.10.12:80 max_fails=2 fail_timeout=10s weight=1;
    
    # Server Cadangan Panas (Hanya aktif jika SEMUA node utama mati!)
    server 192.168.10.13:80 backup;
}

# 2. Server Block Entrypoint Publik
server {
    listen 80;
    server_name portal.smk.lan;

    location / {
        proxy_pass http://web_cluster;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Failover Instan ke Node Berikutnya jika Node Saat Ini Error
        proxy_next_upstream error timeout http_500 http_502 http_503;
        proxy_connect_timeout 2s;
    }
}

4 Laboratorium Troubleshooting: 502 Bad Gateway

# KASUS: 502 Bad Gateway saat mengakses IP Load Balancer # Penyebab: Nginx Load Balancer tidak dapat terhubung ke IP/Port backend node (layanan web di backend mati atau terblokir firewall). # Diagnosa: sudo tail -n 20 /var/log/nginx/error.log # Pesan Error: "connect() failed (111: Connection refused) while connecting to upstream" # Solusi: Pastikan apache/nginx di seluruh node backend berstatus active running dan port 80 terbuka di firewall masing-masing node!
