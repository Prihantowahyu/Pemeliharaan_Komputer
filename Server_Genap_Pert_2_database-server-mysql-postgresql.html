2 Instalasi & Konfigurasi MySQL Server

# 1. Instalasi MySQL Server sudo apt update && sudo apt install mysql-server -y sudo systemctl enable --now mysql # 2. Amankan instalasi (hapus user anonim, nonaktifkan root remote login) sudo mysql_secure_installation # 3. Login ke MySQL shell sebagai root sudo mysql -u root -p # ── Di dalam MySQL Shell ── -- Buat database untuk aplikasi sekolah CREATE DATABASE db_sekolah CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci; -- Buat user khusus aplikasi (bukan root!) CREATE USER 'app_user'@'192.168.10.%' IDENTIFIED BY 'P@ssw0rd!2026'; -- Berikan hak akses hanya ke database tersebut GRANT ALL PRIVILEGES ON db_sekolah.* TO 'app_user'@'192.168.10.%'; FLUSH PRIVILEGES; SHOW GRANTS FOR 'app_user'@'192.168.10.%';
# Izinkan koneksi remote: edit bind-address di /etc/mysql/mysql.conf.d/mysqld.cnf sudo sed -i 's/bind-address\s*=\s*127.0.0.1/bind-address = 0.0.0.0/' /etc/mysql/mysql.conf.d/mysqld.cnf sudo systemctl restart mysql # Uji koneksi dari client (gunakan MySQL Workbench atau CLI) mysql -h 192.168.10.1 -u app_user -p db_sekolah
