1 Hardening SSH — /etc/ssh/sshd_config

# /etc/ssh/sshd_config — Parameter Hardening Enterprise Port 2222 # Ganti port default 22 untuk menghindari scan otomatis PermitRootLogin no # Larang login langsung sebagai root PasswordAuthentication no # Nonaktifkan login password — wajib pakai key PubkeyAuthentication yes # Izinkan autentikasi Public Key MaxAuthTries 3 # Maksimal 3 percobaan login gagal per koneksi LoginGraceTime 30 # Batas waktu login: 30 detik AllowUsers sysadmin guru # Hanya izinkan user tertentu login SSH X11Forwarding no # Nonaktifkan X11 forwarding jika tidak dipakai Banner /etc/ssh/banner.txt # Tampilkan peringatan hukum sebelum login sudo systemctl restart ssh

2 SSH Key-Based Authentication & fail2ban

# Di CLIENT — Generate SSH keypair RSA 4096-bit ssh-keygen -t rsa -b 4096 -C "admin@smk.lan" -f ~/.ssh/id_smk # Salin public key ke server (menggunakan port 2222) ssh-copy-id -i ~/.ssh/id_smk.pub -p 2222 sysadmin@192.168.10.1 # Login tanpa password setelah key terdistribusi ssh -p 2222 -i ~/.ssh/id_smk sysadmin@192.168.10.1
# Di SERVER — Konfigurasi fail2ban untuk SSH brute-force protection sudo apt install fail2ban -y sudo cp /etc/fail2ban/jail.conf /etc/fail2ban/jail.local # Edit /etc/fail2ban/jail.local [sshd] enabled = true port = 2222 logpath = %(sshd_log)s maxretry = 3 bantime = 3600 # Ban 1 jam findtime = 600 # Dalam 10 menit terakhir sudo systemctl enable --now fail2ban sudo fail2ban-client status sshd # Cek status & IP yang di-ban
