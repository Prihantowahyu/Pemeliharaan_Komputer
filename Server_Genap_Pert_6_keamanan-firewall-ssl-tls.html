2 Perintah CLI Firewall UFW & Konfigurasi Virtual Host HTTPS

# 1. Konfigurasi Firewall UFW Standar Zero Trust sudo ufw default deny incoming sudo ufw default allow outgoing sudo ufw allow 2222/tcp comment 'Custom SSH' sudo ufw allow 80/tcp comment 'HTTP Web' sudo ufw allow 443/tcp comment 'HTTPS Secure' sudo ufw enable # 2. Pembuatan Sertifikat SSL/TLS X.509 Mandiri (Self-Signed 365 Hari) sudo openssl req -x509 -nodes -days 365 -newkey rsa:2048 \ -keyout /etc/ssl/private/portal.key \ -out /etc/ssl/certs/portal.crt \ -subj "/C=ID/ST=JawaTengah/L=Semarang/O=SMK TKJ/CN=portal.smk.lan"
# 3. /etc/apache2/sites-available/portal-ssl.conf — Virtual Host HTTPS 
<VirtualHost *:443>
    ServerName portal.smk.lan
    DocumentRoot /var/www/portal

    SSLEngine on
    SSLCertificateFile /etc/ssl/certs/portal.crt
    SSLCertificateKeyFile /etc/ssl/private/portal.key

    # Hardening Protokol Modern (Hanya TLSv1.2 & TLSv1.3)
    SSLProtocol all -SSLv3 -TLSv1 -TLSv1.1
</VirtualHost>

# Redirect Otomatis HTTP (Port 80) ke HTTPS (Port 443)
<VirtualHost *:80>
    ServerName portal.smk.lan
    Redirect permanent / https://portal.smk.lan/
</VirtualHost>
